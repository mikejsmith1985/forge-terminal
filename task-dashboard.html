🔌 API Endpoints

POST /api/routing/classify
Request:  { prompt: string, mentionedFiles: string[] }
Response: { pattern: string, complexity: number, reasoning: string }

POST /api/routing/recommend
Request:  { pattern: string, current_model: string }
Response: { 
  recommended_model: string,
  reasoning: string,
  historical_data: boolean,
  model_comparison: [...],
  cost_comparison: {...}
}

POST /api/routing/log
Request:  { pattern: string, model: string, prompts: number, success: boolean }
Response: { status: "logged" }

GET /api/routing/effectiveness-log
Response: [ { pattern, model, prompts, success, timestamp }, ... ]
