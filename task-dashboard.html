🆕 Conflict Detection Logic

// FollowMeDebugger.jsx - Stream end handler stream.getVideoTracks()[0].onended = () => { if (!streamEndedByUserRef.current) { // Check recent events (within 2 seconds) const recentEvents = eventsRef.current.filter(e => e.timestamp > Date.now() - startTimeRef.current - 2000 ); if (recentEvents.length > 0) { // Tool conflict detected - continue without video console.log('Screen capture tool conflict - continuing'); return; // Don't persist or trigger recovery } // User actually stopped - save for recovery saveSessionToLocalStorage(); } };

🆕 Paste Priority Logic

// ForgeTerminal.jsx - Paste handler const text = e.clipboardData.getData('text/plain'); const hasMedia = Array.from(e.clipboardData.items).some( item => item.type.startsWith('image/') || item.type.startsWith('video/') ); // Prioritize media over text if (hasMedia) { console.log('Media detected, ignoring text path'); // Fall through to media upload handler } else if (text) { // Only send text if NO media present wsRef.current.send(text); return; }
